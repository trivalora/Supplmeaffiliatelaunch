## Technical Implementation

### Script Logic

```javascript
// scripts/generate-component-map.mjs

// 1. Parse routes.config.ts
const routesContent = readFileSync('src/routes.config.ts', 'utf-8');

// 2. Extract components by category
const routePattern = /componentName:\s*['"]([^'"]+)['"][\s\S]*?category:\s*['"]([^'"]+)['"]/g;

while ((match = routePattern.exec(routesArrayContent)) !== null) {
  if (category === 'knowledgebase') supplementRoutes.push(componentName);
  if (category === 'comparison') comparisonRoutes.push(componentName);
}

// 3. Generate imports
const supplementImports = supplementRoutes.map(name => 
  `import { ${name} } from '@/components/pages/supplements/${name}';`
);

const comparisonImports = comparisonRoutes.join(',\n  ');

// 4. Generate component map
const mapEntries = [...supplementRoutes, ...comparisonRoutes].map(name =>
  `  '${name}': ${name},`
);

// 5. Write to app/[slug]/page.tsx
writeFileSync('app/[slug]/page.tsx', generatedContent);
```

### Build Integration

```json
// package.json

  "scripts": {
    "generate:component-map": "node scripts/generate-component-map.mjs",
    "prebuild": "npm run generate:component-map",  // Auto-runs before build
    "build": "next build"
  }
}
```
