  if (row.overview_content) populated++;
  
  console.log(row.slug.padEnd(20), overview.padEnd(8), additional);
});

console.log('\n' + '='.repeat(50));
console.log(`✅ Content populated: ${populated}/17 supplements`);

import { createClient } from '@/lib/supabase/server';
import { KnowledgebaseTemplate } from '@/components/templates/KnowledgebaseTemplate';

export async function AshwagandhaKnowledgebasePage() {

### Step 5: Verify Autolinking

**Before migration**: 31 glossary terms have 0 incoming links

**After migration**: All 31 should have links from knowledgebase pages

**Check autolinking** (`scripts/verify-autolinking.mjs`):

```javascript
#!/usr/bin/env node
import { createClient } from '@supabase/supabase-js';
import { config } from 'dotenv';

config({ path: '.env.local' });

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL,
  process.env.SUPABASE_SERVICE_ROLE_KEY,
  { db: { schema: 'api' } }
);

// Fetch all supplements with content
const { data: supplements } = await supabase
  .from('supplements')
  .select('slug, overview_content, additional_overview_content');

// Fetch all glossary terms
const { data: terms } = await supabase
  .from('glossary_terms')
  .select('slug, term');

console.log('\n📊 Autolinking Verification:\n');
console.log('Term'.padEnd(30), 'Found In', 'Count');
console.log('-'.repeat(60));

let totalLinks = 0;
terms.forEach(term => {
  const matches = [];
  const regex = new RegExp(`\\b${term.term}\\b`, 'gi');
  
  supplements.forEach(supp => {
    const content = `${supp.overview_content || ''} ${supp.additional_overview_content || ''}`;
    const count = (content.match(regex) || []).length;
    if (count > 0) {
      matches.push(`${supp.slug} (${count})`);
      totalLinks += count;
    }
  });
  
  if (matches.length > 0) {
    console.log(term.term.padEnd(30), matches.join(', '));
  }
});

console.log('\n' + '='.repeat(60));
console.log(`✅ Total autolinkable terms: ${totalLinks}`);
console.log('🎉 Autolinking ready!');
```

**Expected**: 31+ autolinkable instances found

## 🐛 Troubleshooting

### Issue: "Cannot find module '@/lib/supabase/server'"

**Solution**: Check that `lib/supabase/server.ts` exists with:

```typescript
import { createServerClient } from '@supabase/ssr';
import { cookies } from 'next/headers';

export function createClient() {
  const cookieStore = cookies();
  
  return createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        get(name: string) {
          return cookieStore.get(name)?.value;
        },
      },
      db: { schema: 'api' },
    }
  );
}
```

### Issue: "Type 'Promise<JSX.Element>' is not assignable"

**Solution**: Component must be async:

```tsx
// ❌ WRONG
export function AshwagandhaKnowledgebasePage() { ... }

// ✅ CORRECT
export async function AshwagandhaKnowledgebasePage() { ... }
```

### Issue: Content not displaying

**Solution**: Check database query returns data:

```tsx
const { data: supplement, error } = await supabase
  .from('supplements')
  .select('overview_content, additional_overview_content')
  .eq('slug', 'ashwagandha')
  .single();

if (error) {
  console.error('Database error:', error);
}

if (!supplement) {
  console.error('No supplement found with slug: ashwagandha');
}
```

### Issue: Autolinking not working

**Solution**: Verify `currentPage` prop is passed:

```tsx
const pageProps = {
  supplementName: "Ashwagandha",
  currentPage: "ashwagandha", // ✅ REQUIRED for self-link prevention
  overviewContent: supplement?.overview_content,
  // ...
};
```
